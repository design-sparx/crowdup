**Create `src/lib/theme.ts`:**

```ts
export const THEME_STORAGE_KEY = "theme";

/** What the page should look like: a stored light/dark choice wins, anything else follows the system. */
export function resolveTheme(stored: string | null, prefersDark: boolean): "light" | "dark" {
    if (stored === "light" || stored === "dark") return stored;
    return prefersDark ? "dark" : "light";
}

/**
 * Inline script for <head>. It runs before first paint so there is no flash of the wrong theme.
 * Keep the logic identical to resolveTheme above.
 */
export const themeInitScript = `(function(){try{var s=localStorage.getItem("${THEME_STORAGE_KEY}");var d=window.matchMedia("(prefers-color-scheme: dark)").matches;var dark=s==="dark"||(s!=="light"&&d);document.documentElement.classList.toggle("dark",dark);}catch(e){}})();`;
```

```ts
export interface Account {
    name: string;
    avatar: string | null;

export function initials(name: string): string {
    const words = name.trim().split(/\s+/).filter(Boolean);
```

### C1. `theme-toggle/`

`theme-toggle.tsx` (stateless on purpose: the icon swap is pure CSS, so there is no hydration mismatch):

```tsx
import {IconMoon, IconSun} from "@tabler/icons-react";
import {Button} from "@/components/ui/button";
import {THEME_STORAGE_KEY} from "@/lib/theme";

function toggleTheme() {
    const root = document.documentElement;
    const dark = !root.classList.contains("dark");
    root.classList.toggle("dark", dark);
    try {
        localStorage.setItem(THEME_STORAGE_KEY, dark ? "dark" : "light");
    } catch {
        // Storage can be blocked (private mode); the toggle still works for this page view.
    }
}

export function ThemeToggle() {
    return (
        <Button variant="ghost" size="icon" aria-label="Toggle theme" onClick={toggleTheme}>
            <IconSun className="hidden dark:block"/>
            <IconMoon className="dark:hidden"/>
        </Button>
    );
}
```

`theme-toggle.test.tsx`:

```tsx

    document.documentElement.classList.remove("dark");
    localStorage.clear();
});

    it("switches to dark, remembers it, and switches back", async () => {
        render(<ThemeToggle/>);
        const button = screen.getByRole("button", {name: "Toggle theme"});
```

```tsx
// @vitest-environment jsdom
import {describe, expect, it, vi} from "vitest";
import {render, screen} from "@testing-library/react";
import {createMemoryRouter, RouterProvider} from "react-router";
import {ErrorPage} from "./error-page";

function Boom({response}: {response: Response | Error}): never {
    throw response;
}

function renderError(thrown: Response | Error) {
    const router = createMemoryRouter([
        {path: "/", element: <Boom response={thrown}/>, errorElement: <ErrorPage/>},
    ]);
    render(<RouterProvider router={router}/>);
}

describe("ErrorPage", () => {
    it("says page not found for a 404 response", async () => {
        renderError(new Response(null, {status: 404}));
        expect(await screen.findByRole("heading", {name: "Page not found"})).toBeInTheDocument();
        expect(screen.getByRole("link", {name: "Back to home"})).toHaveAttribute("href", "/");
    });
    it("says something went wrong for any other error", async () => {
        vi.spyOn(console, "error").mockImplementation(() => {});
        renderError(new Error("boom"));
        expect(await screen.findByRole("heading", {name: "Something went wrong"})).toBeInTheDocument();
    });
});
```

```ts
    const account: Account | null = user

}
```
